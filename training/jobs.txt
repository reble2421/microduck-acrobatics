# このリポジトリの方策を作った学習の一覧。上から順に1本ずつ（bash training/night_queue.sh training/jobs.txt）。
# 書式: タスクID [追加オプション]    （# から後ろはコメント）
# どれも RTX 4060 Laptop、2048体の並列、1反復 約2.7〜3.3秒。

# 0) 土台：逆立ちと後転を最初から 10,000反復（最終時点 model_9999）
#    -> policies/headstand.onnx（逆立ち）、policies/backroll.onnx（後転、この課題の前に学習した前作）
#    逆立ちの公開版の設定は、ヘッドスピンの土台として整理したもので、実際の学習の設定とは、指令の範囲と
#    それに関わる報酬が一部異なる（ヘッドスピン以降の学習の設定が、実際に使ったものと同じであることは確認済み）。
#    手元に model_9999 がある場合は、この2行を消して 1) から回す。
Mjlab-HeadStand-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 10000
Mjlab-BackRoll-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 10000

# 1) ヘッドスピン：逆立ち（model_9999）の続きから 3,000反復 -> policies/headspin.onnx
Mjlab-HeadSpin-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 3000 --agent.resume True --agent.load-run .*_microduck_headstand --agent.load-checkpoint model_9999.pt --agent.run-name headspin_from_v1

# 2) 遅れに強いヘッドスピン：1) の最終時点（model_12998）の続きから 2,000反復 -> policies/headspin_robust.onnx
Mjlab-HeadSpinRobust-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 2000 --agent.resume True --agent.load-run .*_headspin_from_v1 --agent.load-checkpoint model_12998.pt --agent.run-name headspin_robust_v1

# 3) 遅れに強い後転：後転（model_9999）の続きから 3,000反復 -> policies/backroll_robust.onnx
Mjlab-BackRollRobust-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 3000 --agent.resume True --agent.load-run .*_microduck_backroll --agent.load-checkpoint model_9999.pt --agent.run-name backroll_robust

# 4) 速さの限界を探る追加の学習：2) の最終時点（model_14997）の続きから 各2,500反復
#    -> policies/headspin_fast6.onnx, policies/headspin_fast5.onnx（連続技には使っていない）
Mjlab-HeadSpinFast6-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 2500 --agent.resume True --agent.load-run .*_headspin_robust_v1 --agent.load-checkpoint model_14997.pt --agent.run-name headspin_fast6
Mjlab-HeadSpinFast5-Flat-MicroDuck --env.scene.num-envs 2048 --agent.max-iterations 2500 --agent.resume True --agent.load-run .*_headspin_robust_v1 --agent.load-checkpoint model_14997.pt --agent.run-name headspin_fast5

# 5) 比較：1) と同じヘッドスピンの設定を、逆立ちの続きではなく最初から 8,000反復（初期化の比較。方策は載せていない）
#    結果は results/compare/headspin_init_delaytrain（10回中0回）、学習の推移は results/training/headspin_curves.csv
